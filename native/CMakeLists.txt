cmake_minimum_required(VERSION 3.24)
project(ScreenshotOpenCvSharp LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This recipe supports MSVC Windows x64 only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(OpenCV_STATIC ON)
find_package(OpenCV 4.13 REQUIRED COMPONENTS core imgproc imgcodecs)
# Keep upstream bindings unmodified; feature guards remove unsupported module bodies.
file(GLOB binding_sources CONFIGURE_DEPENDS "${SCREENSHOT_OPENCVSHARP_SOURCE}/src/OpenCvSharpExtern/*.cpp")
if(NOT binding_sources)
  message(FATAL_ERROR "Pinned OpenCvSharp sources unavailable")
endif()
add_library(OpenCvSharpExtern SHARED ${binding_sources})
target_include_directories(OpenCvSharpExtern PRIVATE ${OpenCV_INCLUDE_DIRS})
target_compile_definitions(OpenCvSharpExtern PRIVATE NO_CONTRIB NO_STITCHING NO_CALIB3D NO_VIDEO NO_FEATURES2D NO_FLANN NO_DNN NO_ML NO_OBJDETECT NO_PHOTO NO_BARCODE NO_HIGHGUI NO_VIDEOIO NO_ICONV)
target_compile_options(OpenCvSharpExtern PRIVATE /O2 /GS /guard:cf)
target_link_libraries(OpenCvSharpExtern PRIVATE ${OpenCV_LIBS})
# The map and library-resolution log are evidence of this build, not a vendor inference.
target_link_options(OpenCvSharpExtern PRIVATE /guard:cf /DYNAMICBASE /HIGHENTROPYVA /NXCOMPAT /Brepro "/MAP:${CMAKE_BINARY_DIR}/OpenCvSharpExtern.map" /MAPINFO:EXPORTS /VERBOSE:LIB)
